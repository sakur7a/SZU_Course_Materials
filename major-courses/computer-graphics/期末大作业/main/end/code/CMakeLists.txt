cmake_minimum_required(VERSION 3.16)

# 提交包中的源码可以独立配置；用户也可以显式传入 CMake 工具链。
if(DEFINED ENV{VCPKG_ROOT} AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
endif()
if(DEFINED ENV{VCPKG_DEFAULT_TRIPLET} AND NOT DEFINED VCPKG_TARGET_TRIPLET)
    set(VCPKG_TARGET_TRIPLET "$ENV{VCPKG_DEFAULT_TRIPLET}" CACHE STRING "")
endif()
project(szu_yuehai_campus LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(glad CONFIG REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
add_executable(szu-campus main.cpp InitShader.cpp)
file(GLOB_RECURSE campus_resources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*")
set_property(TARGET szu-campus APPEND PROPERTY LINK_DEPENDS ${campus_resources})
target_include_directories(szu-campus PRIVATE include)
target_link_libraries(szu-campus PRIVATE glad::glad glfw glm::glm)
if(MSVC)
    target_compile_options(szu-campus PRIVATE /utf-8)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(szu-campus PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
if(MINGW)
    target_link_options(szu-campus PRIVATE -static -static-libgcc -static-libstdc++)
endif()
foreach(resource shaders assets)
    add_custom_command(TARGET szu-campus POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/${resource}" "$<TARGET_FILE_DIR:szu-campus>/${resource}")
endforeach()
