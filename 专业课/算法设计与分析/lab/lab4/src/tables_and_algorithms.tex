\section*{算法伪代码}

\subsection*{算法1：基础动态规划}
\begin{algorithm}[H]
\caption{Basic DP: $O(e \cdot f^2)$ time, $O(e \cdot f)$ space}
\begin{algorithmic}[1]
\REQUIRE eggs $e$, floors $f$
\ENSURE minimum trials
\IF{$e \leq 0$ or $f \leq 0$} \RETURN 0 \ENDIF
\IF{$e = 1$} \RETURN $f$ \ENDIF
\STATE $dp[1..e][0..f] \leftarrow 0$
\FOR{$j \leftarrow 1$ to $f$}
    \STATE $dp[1][j] \leftarrow j$
\ENDFOR
\FOR{$i \leftarrow 2$ to $e$}
    \FOR{$j \leftarrow 2$ to $f$}
        \STATE $dp[i][j] \leftarrow \infty$
        \FOR{$x \leftarrow 1$ to $j$}
            \STATE $dp[i][j] \leftarrow \min(dp[i][j],\ 1 + \max(dp[i-1][x-1],\ dp[i][j-x]))$
        \ENDFOR
    \ENDFOR
\ENDFOR
\RETURN $dp[e][f]$
\end{algorithmic}
\end{algorithm}

\subsection*{算法2：二分查找优化DP}
\begin{algorithm}[H]
\caption{Binary Search DP: $O(e \cdot f \cdot \log f)$ time, $O(e \cdot f)$ space}
\begin{algorithmic}[1]
\REQUIRE eggs $e$, floors $f$
\ENSURE minimum trials
\IF{$e \leq 0$ or $f \leq 0$} \RETURN 0 \ENDIF
\IF{$e = 1$} \RETURN $f$ \ENDIF
\FOR{$i \leftarrow 2$ to $e$}
    \FOR{$j \leftarrow 2$ to $f$}
        \STATE $dp[i][j] \leftarrow \infty$, $lo \leftarrow 1$, $hi \leftarrow j$
        \WHILE{$lo \leq hi$}
            \STATE $mid \leftarrow \lfloor(lo + hi) / 2\rfloor$
            \STATE $cost \leftarrow 1 + \max(dp[i-1][mid-1],\ dp[i][j-mid])$
            \STATE $dp[i][j] \leftarrow \min(dp[i][j],\ cost)$
            \IF{$dp[i-1][mid-1] < dp[i][j-mid]$}
                \STATE $lo \leftarrow mid + 1$
            \ELSE
                \STATE $hi \leftarrow mid - 1$
            \ENDIF
        \ENDWHILE
    \ENDFOR
\ENDFOR
\RETURN $dp[e][f]$
\end{algorithmic}
\end{algorithm}

\subsection*{算法3：交替DP（最优解法）}
\begin{algorithm}[H]
\caption{Alternative DP: $O(e \cdot \mathrm{answer})$ time, $O(e)$ space}
\begin{algorithmic}[1]
\REQUIRE eggs $e$, floors $f$
\ENSURE minimum trials
\IF{$e \leq 0$ or $f \leq 0$} \RETURN 0 \ENDIF
\IF{$e = 1$} \RETURN $f$ \ENDIF
\STATE $dp[0..e] \leftarrow 0$, $trials \leftarrow 0$
\WHILE{$dp[e] < f$}
    \STATE $trials \leftarrow trials + 1$
    \FOR{$j \leftarrow e$ downto $1$}
        \STATE $dp[j] \leftarrow dp[j] + dp[j-1] + 1$
    \ENDFOR
\ENDWHILE
\RETURN $trials$
\end{algorithmic}
\end{algorithm}

\subsection*{算法4：组合数方法}
\begin{algorithm}[H]
\caption{Binomial Method: $O(\mathrm{answer} \cdot \min(e, \mathrm{answer}))$ time, $O(1)$ space}
\begin{algorithmic}[1]
\REQUIRE eggs $e$, floors $f$
\ENSURE minimum trials
\IF{$e \leq 0$ or $f \leq 0$} \RETURN 0 \ENDIF
\IF{$e = 1$} \RETURN $f$ \ENDIF
\STATE $n \leftarrow 1$
\WHILE{true}
    \STATE $total \leftarrow 0$
    \FOR{$k \leftarrow 1$ to $\min(e, n)$}
        \STATE $total \leftarrow total + \binom{n}{k}$
        \IF{$total \geq f$} \RETURN $n$ \ENDIF
    \ENDFOR
    \STATE $n \leftarrow n + 1$
\ENDWHILE
\end{algorithmic}
\end{algorithm}
