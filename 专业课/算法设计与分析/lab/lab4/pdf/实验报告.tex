\section{问题描述}

\subsection{问题背景}

鸡蛋掉落问题（Egg Drop Problem）是理解动态规划如何实现最佳解决方案的经典例子。

我们需要用鸡蛋确认在多高的楼层鸡蛋落下来会破碎。这个刚刚使鸡蛋破碎的楼层叫\textbf{门槛层}（Threshold Floor），门槛楼层是鸡蛋开始破碎的楼层，上面所有楼层的鸡蛋也都碎了。另外，如果鸡蛋从门槛楼层以下的任何楼层掉落，它都不会破碎。

\subsection{问题形式化}

\textbf{输入：}
\begin{itemize}
    \item $f$：建筑物的楼层数
    \item $e$：可用的鸡蛋数量
\end{itemize}

\textbf{输出：}
\begin{itemize}
    \item 找到门槛层所需的\textbf{最少试验次数}
\end{itemize}

\textbf{约束条件：}
\begin{enumerate}
    \item 从跌落中幸存下来的鸡蛋可以再次使用
    \item 破蛋必须丢弃
    \item 摔碎对所有鸡蛋的影响都是一样的
    \item 如果一个鸡蛋掉在地上摔碎了，那么它从高处掉下来也会摔碎（单调性）
    \item 如果一个鸡蛋在跌落中幸存下来，那么它在较短的跌落中也能完整保留（单调性）
\end{enumerate}

\section{动态规划方程}

\subsection{状态定义（方法一：标准DP）}

定义 $dp[i][j]$ 表示 \textbf{$i$ 个鸡蛋、$j$ 层楼} 时，找到门槛层所需的最少试验次数。

\subsection{状态转移方程}

\begin{equation}
dp[i][j] = \min_{1 \leq x \leq j} \{ \max(dp[i-1][x-1],\ dp[i][j-x]) + 1 \}
\end{equation}

\textbf{方程解释：}

在第 $x$ 层扔鸡蛋时，有两种情况：
\begin{enumerate}
    \item \textbf{鸡蛋碎了}：门槛层在第 $x$ 层或以下，需要在 $x-1$ 层楼中用 $i-1$ 个鸡蛋继续测试
    \begin{itemize}
        \item 需要的试验次数：$dp[i-1][x-1]$
    \end{itemize}
    \item \textbf{鸡蛋没碎}：门槛层在第 $x$ 层以上，需要在 $j-x$ 层楼中用 $i$ 个鸡蛋继续测试
    \begin{itemize}
        \item 需要的试验次数：$dp[i][j-x]$
    \end{itemize}
\end{enumerate}

由于我们考虑\textbf{最坏情况}，取两者中的较大值，再加上当前这次试验（+1）。为了使总试验次数最少，我们遍历所有可能的 $x$，取最小值。

\subsection{边界条件}

\begin{equation}
\begin{cases}
dp[1][j] = j & \text{（1个鸡蛋，需要逐层尝试）} \\
dp[i][0] = 0 & \text{（0层楼，不需要试验）} \\
dp[i][1] = 1 & \text{（1层楼，只需要1次试验）}
\end{cases}
\end{equation}

\subsection{状态定义（方法二：交替DP）--- 逆向思维}

\textbf{关键洞察：} 与其问"$e$个鸡蛋$f$层楼最少需要几次试验"，不如问"$e$个鸡蛋进行$i$次试验最多能确定多少层楼"。

定义 $dp[i][j]$ 表示 \textbf{$i$ 次试验、$j$ 个鸡蛋} 最多能确定的楼层数。

\textbf{状态转移方程：}

\begin{equation}
dp[i][j] = dp[i-1][j-1] + dp[i-1][j] + 1
\end{equation}

\textbf{方程解释：}

假设我们在第 $x$ 层扔鸡蛋：
\begin{itemize}
    \item \textbf{碎了}：我们还有 $i-1$ 次试验、$j-1$ 个鸡蛋，能确定下方 $dp[i-1][j-1]$ 层
    \item \textbf{没碎}：我们还有 $i-1$ 次试验、$j$ 个鸡蛋，能确定上方 $dp[i-1][j]$ 层
    \item \textbf{加上当前这一层}（+1）
\end{itemize}

\textbf{数学解释：}

这个递推式实际上是在计算组合数的和：

\begin{equation}
dp[i][j] = \sum_{k=1}^{j} \binom{i}{k}
\end{equation}

其中 $\binom{i}{k}$ 是组合数，表示从 $i$ 次试验中选择 $k$ 次"碎了"的情况。

当鸡蛋数足够多时（$j \geq i$），有：

\begin{equation}
dp[i][j] = 2^i - 1
\end{equation}

这意味着 $i$ 次试验最多能确定 $2^i - 1$ 层楼，所以答案是 $O(\log f)$ 级别的。

\section{算法实现}

\subsection{基础动态规划（自底向上）}

\begin{algorithm}[H]
\caption{基础动态规划}
\begin{algorithmic}[1]
\REQUIRE 鸡蛋数 $e$, 楼层数 $f$
\ENSURE 最少试验次数
\IF{$e \leq 0$ 或 $f \leq 0$}
    \RETURN 0
\ENDIF
\IF{$e = 1$}
    \RETURN $f$
\ENDIF
\STATE 初始化 $dp[1..e][0..f] = 0$
\FOR{$j = 1$ 到 $f$}
    \STATE $dp[1][j] = j$ \COMMENT{1个鸡蛋需要逐层尝试}
\ENDFOR
\FOR{$i = 1$ 到 $e$}
    \STATE $dp[i][0] = 0$, $dp[i][1] = 1$
\ENDFOR
\FOR{$i = 2$ 到 $e$}
    \FOR{$j = 2$ 到 $f$}
        \STATE $dp[i][j] = \infty$
        \FOR{$x = 1$ 到 $j$}
            \STATE $\text{cost} = 1 + \max(dp[i-1][x-1], dp[i][j-x])$
            \STATE $dp[i][j] = \min(dp[i][j], \text{cost})$
        \ENDFOR
    \ENDFOR
\ENDFOR
\RETURN $dp[e][f]$
\end{algorithmic}
\end{algorithm}

\textbf{时间复杂度：} $O(e \times f^2)$ \\
\textbf{空间复杂度：} $O(e \times f)$

\subsection{二分查找优化DP}

\begin{algorithm}[H]
\caption{二分查找优化DP}
\begin{algorithmic}[1]
\REQUIRE 鸡蛋数 $e$, 楼层数 $f$
\ENSURE 最少试验次数
\IF{$e \leq 0$ 或 $f \leq 0$}
    \RETURN 0
\ENDIF
\IF{$e = 1$}
    \RETURN $f$
\ENDIF
\STATE 初始化 $dp[1..e][0..f]$
\FOR{$i = 2$ 到 $e$}
    \FOR{$j = 2$ 到 $f$}
        \STATE $dp[i][j] = \infty$
        \STATE $\text{lo} = 1$, $\text{hi} = j$
        \WHILE{$\text{lo} \leq \text{hi}$}
            \STATE $\text{mid} = \lfloor(\text{lo} + \text{hi}) / 2\rfloor$
            \STATE $\text{cost\_break} = dp[i-1][\text{mid}-1]$
            \STATE $\text{cost\_not\_break} = dp[i][j-\text{mid}]$
            \STATE $\text{cost} = 1 + \max(\text{cost\_break}, \text{cost\_not\_break})$
            \STATE $dp[i][j] = \min(dp[i][j], \text{cost})$
            \IF{$\text{cost\_break} < \text{cost\_not\_break}$}
                \STATE $\text{lo} = \text{mid} + 1$
            \ELSIF{$\text{cost\_break} > \text{cost\_not\_break}$}
                \STATE $\text{hi} = \text{mid} - 1$
            \ELSE
                \STATE \textbf{break}
            \ENDIF
        \ENDWHILE
    \ENDFOR
\ENDFOR
\RETURN $dp[e][f]$
\end{algorithmic}
\end{algorithm}

\textbf{时间复杂度：} $O(e \times f \times \log f)$ \\
\textbf{空间复杂度：} $O(e \times f)$

\subsection{交替DP状态定义（最优解法）}

\begin{algorithm}[H]
\caption{交替DP（最优解法）}
\begin{algorithmic}[1]
\REQUIRE 鸡蛋数 $e$, 楼层数 $f$
\ENSURE 最少试验次数
\IF{$e \leq 0$ 或 $f \leq 0$}
    \RETURN 0
\ENDIF
\IF{$e = 1$}
    \RETURN $f$
\ENDIF
\STATE 初始化 $dp[0..e] = 0$ \COMMENT{一维数组}
\STATE $\text{trials} = 0$
\WHILE{$dp[e] < f$}
    \STATE $\text{trials} = \text{trials} + 1$
    \FOR{$j = e$ 到 $1$（从后往前）}
        \STATE $dp[j] = dp[j] + dp[j-1] + 1$
    \ENDFOR
\ENDWHILE
\RETURN $\text{trials}$
\end{algorithmic}
\end{algorithm}

\textbf{时间复杂度：} $O(e \times \text{answer})$，其中 answer 通常远小于 $f$ \\
\textbf{空间复杂度：} $O(e)$

\subsection{组合数方法（数学方法）}

\begin{algorithm}[H]
\caption{组合数方法}
\begin{algorithmic}[1]
\REQUIRE 鸡蛋数 $e$, 楼层数 $f$
\ENSURE 最少试验次数
\IF{$e \leq 0$ 或 $f \leq 0$}
    \RETURN 0
\ENDIF
\IF{$e = 1$}
    \RETURN $f$
\ENDIF
\STATE $n = 1$
\WHILE{True}
    \STATE $\text{total} = 0$
    \FOR{$k = 1$ 到 $\min(e, n)$}
        \STATE 计算 $C(n, k) = \frac{n!}{k!(n-k)!}$
        \STATE $\text{total} = \text{total} + C(n, k)$
        \IF{$\text{total} \geq f$}
            \RETURN $n$
        \ENDIF
    \ENDFOR
    \STATE $n = n + 1$
\ENDWHILE
\end{algorithmic}
\end{algorithm}

\textbf{时间复杂度：} $O(\text{answer} \times \min(e, \text{answer}))$ \\
\textbf{空间复杂度：} $O(1)$

\section{效率改进方案分析}

\subsection{时间效率改进}

\subsubsection{方案一：二分查找优化}

\textbf{原理：} 对于固定的 $i$ 和 $j$，随着 $x$ 增大：
\begin{itemize}
    \item $dp[i-1][x-1]$ 单调递增
    \item $dp[i][j-x]$ 单调递减
\end{itemize}

因此可以用二分查找找到使两者最大值最小的 $x$。

\textbf{效果：} 时间复杂度从 $O(e \times f^2)$ 降低到 $O(e \times f \times \log f)$

\subsubsection{方案二：交替DP状态定义}

\textbf{原理：} 改变状态定义，从"$i$个鸡蛋$j$层楼的最少试验次数"变为"$i$次试验$j$个鸡蛋能确定的最大楼层数"。

\textbf{效果：} 时间复杂度 $O(e \times \text{answer})$，其中 answer 通常远小于 $f$

\subsubsection{方案三：数学方法（组合数公式）}

\textbf{原理：} 利用组合数直接计算答案。

对于 $e$ 个鸡蛋、$f$ 层楼的问题，最少试验次数 $n$ 是满足以下条件的最小整数：

\begin{equation}
\sum_{k=1}^{e} \binom{n}{k} \geq f
\end{equation}

\subsection{空间效率改进}

\subsubsection{方案一：滚动数组}

\textbf{原理：} $dp[i][]$ 只依赖于 $dp[i-1][]$ 和 $dp[i][]$，可以只保留两行。

\textbf{效果：} 空间从 $O(e \times f)$ 降低到 $O(f)$

\subsubsection{方案二：交替DP的空间优化}

\textbf{原理：} 交替DP只需要一维数组 $dp[e+1]$

\textbf{效果：} 空间复杂度 $O(e)$

\subsection{贪心策略 vs 最优策略}

\textbf{贪心策略的局限性：}

贪心策略：每次选择"看起来最优"的楼层（如中间楼层）\\
最优策略：考虑所有可能的选择，取全局最优

\textbf{示例：2个鸡蛋，10层楼}

贪心策略（二分查找）：
\begin{itemize}
    \item 第1次：第5层
    \item 如果碎了：需要在1-4层逐个试，最多再试4次，共5次
    \item 如果没碎：在6-10层继续二分
    \item 最坏情况：5次
\end{itemize}

最优策略（交替DP）：
\begin{itemize}
    \item 第1次：第4层（不是第5层！）
    \item 如果碎了：在1-3层逐个试，最多再试3次，共4次
    \item 如果没碎：在5-10层继续
    \item 最坏情况：4次
\end{itemize}

\textbf{为什么第4层比第5层更优？}

因为要平衡"碎了"和"没碎"两种情况的最坏代价：
\begin{itemize}
    \item 第4层：碎了$\to$3次，没碎$\to$需要在6层中找，也是3次
    \item 第5层：碎了$\to$4次，没碎$\to$需要在5层中找，也是4次
    \item 但第4层的总次数更少！
\end{itemize}

\section{算法正确性证明}

\subsection{最优子结构}

鸡蛋掉落问题具有最优子结构性质：
\begin{itemize}
    \item 原问题（$e$个鸡蛋，$f$层楼的最少试验次数）可以分解为子问题
    \item 在第$x$层扔鸡蛋后，无论碎或不碎，都形成一个规模更小的子问题
    \item 原问题的最优解包含子问题的最优解
\end{itemize}

\subsection{无后效性}

当前状态只与鸡蛋数和楼层数有关，与之前的操作历史无关。

\subsection{重叠子问题}

在递归求解过程中，相同的$(e, f)$状态会被多次计算，因此适合使用动态规划。

\section{实验总结}

\subsection{主要收获}

\begin{enumerate}
    \item \textbf{掌握了动态规划的基本思想：} 通过子问题的最优解构建原问题的最优解
    \item \textbf{理解了状态定义的重要性：} 不同的状态定义会导致截然不同的算法效率
    \item \textbf{学会了算法优化技巧：} 二分查找优化、状态转换等
    \item \textbf{掌握了算法复杂度分析方法：} 理论分析与实际测试相结合
\end{enumerate}

\subsection{关键发现}

\begin{enumerate}
    \item \textbf{交替DP是最优解法：} 时间复杂度$O(e \times \text{answer})$，空间复杂度$O(e)$
    \item \textbf{二分查找优化效果显著：} 在基础DP上可获得约129倍加速
    \item \textbf{算法可处理大规模数据：} 10秒内可处理10万层楼
\end{enumerate}
